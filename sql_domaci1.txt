select 
stadium_name,stadium_capacity,attendance, stadium_capacity - attendance as empty_seats
from matches_flat  
order by empty_seats desc limit 5 --prve 3 koje imaju najveci broj praznis sedista se odigrao na National Stadium

--2
select 
distinct referee_name, home_team
from matches_flat
where home_team = 'Riverton Athletic' --15

--3
select 
home_team, home_goals_ht,away_goals_ht, home_goals, away_goals, away_team
from matches_flat
where (home_goals_ht < away_goals_ht) and (home_goals > away_goals) --15

--4
select 
home_team, away_team, home_goals_ht, away_goals_ht, home_goals, away_goals, home_goals +away_goals as FT, home_goals_ht + away_goals_ht as HT,
(home_goals +away_goals) -( home_goals_ht + away_goals_ht) as goals_second_half , ( home_goals_ht + away_goals_ht) as goals_first_half
from matches_flat
where ((home_goals +away_goals) -( home_goals_ht + away_goals_ht)) > ( home_goals_ht + away_goals_ht) 
order by FT desc limit 1  -- Northport United-Newbury Forest reza. na poluvremenu  1:0 , rez na kraju 6:3

--5
select 
season, weather
from matches_flat
where season ='2023/24' and weather in ('snow', 'fog') --25 utakmica

order by kickoff_ts -- prvi mec pobedio Northport United, drugi je neresen, treci je pobedio Northport City i cetvrti Northport United
